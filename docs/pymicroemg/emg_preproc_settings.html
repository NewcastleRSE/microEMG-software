140    def add_butterworth_filter(
141        self,
142        cutoff_freq: None | list[float] | float = None,
143        order: int = 6,
144        filter_type: str = "bandpass",
145        apply_filter: bool = True,
146    ):
147        """
148        Add filter and settings for a zero-phase Butterworth filter. See
149        scipy.signal.butter for parameter details.
150
151        Parameters
152        ----------
153        cutoff_freq : None | list[float] | float, optional
154            Cutoff frequency or frequencies to pass to scipy.signal.butter. The default
155            is [500, 2000] if the filter type is "bandpass". The cutoff frequency must
156            be specified for other filter types.
157        order : int, optional
158            Filter order; must be even. The default is 6.
159        filter_type : str, optional
160            Type of filter - must be "low", "lowpass", "high", "highpass", or
161            "bandpass". The default is "bandpass".
162        apply_filter: bool, optional
163            Whether to apply the specified filter settings during preprocessing. If
164            False, the specified filter settings will be stored, but not used during
165            preprocessing. This behaviour may be useful when filtering is turned off,
166            but may be turned on later (e.g., by a GUI). The default is True.
167
168        Raises
169        ------
170        ValueError
171            Raised if filter order is not even.
172            Raised if filter type is not one of the allowed types.
173            Raised if filter cutoff frequency is not a list or float.
174            Raised if filter cutoff frequency list has length greater than 2.
175        Exception
176            Raised if cutoff frequency is not provided when the filter type is not
177            bandpass.
178
179        Returns
180        -------
181        None.
182
183        """
184
185        if order % 2 != 0:
186            raise ValueError("The filter order must be an even integer.")
187
188        # Check and format filter type string
189        filter_type = filter_type.lower()
190        if filter_type == "low" or filter_type == "high":
191            filter_type = filter_type + "pass"
192
193        filter_types_allowed = self._get_filter_types_allowed()
194        if filter_type not in filter_types_allowed:
195            raise ValueError(
196                "The filter type must be one of the following: " f"{filter_types_allowed}"
197            )
198
199        # Default cutoff frequencies - only for bandpass filter
200        if cutoff_freq is None:
201            if filter_type == "bandpass":
202                cutoff_freq = [100, 2000]
203            else:
204                raise Exception(
205                    "Cutoff frequencies cutoff_freq must be specified if "
206                    "filter_type is not bandpass"
207                )
208
209        # Split cutoff frequencies so easier to set in GUI
210        if isinstance(cutoff_freq, list):
211            # Always store as float to facilitate tests with GUI
212            # (easier to check consistency)
213            cutoff_freq = [float(i) for i in cutoff_freq]
214
215            # First cutoff frequency
216            cutoff1 = cutoff_freq[0]
217
218            # Check for second cutoff frequency
219            if len(cutoff_freq) == 2:
220                cutoff2 = cutoff_freq[1]
221            elif len(cutoff_freq) > 2:
222                raise ValueError(
223                    "cutoff_freq must be length 1 (for lowpass or highpass filter) or 2"
224                    " (for bandpass filter)"
225                )
226                # Note: Filter function handles checks for filter type and number of
227                # cutoff (critical) frequencies
228            else:
229                cutoff2 = None
230
231        elif isinstance(cutoff_freq, int | float):
232            # Only one cutoff frequency; store as float
233            cutoff1 = float(cutoff_freq)
234            cutoff2 = None
235        else:
236            raise ValueError("cutoff_freq must be a list or float")
237
238        # Save filter settings
239        self.butterworth_filter = apply_filter
240        self.butterworth_filter_settings = {
241            "cutoff1": cutoff1,
242            "cutoff2": cutoff2,
243            "order": order,
244            "filter_type": filter_type,
245        }
246
247    def get_butterworth_filter_cutoff(self) -> float | list[float]:
248        """
249        Extracts the cutoff frequency or frequencies as a single variable (in format
250        suitable to be passed to a filtering function).
251
252        Returns
253        -------
254        float | list[float]
255            Cutoff frequency or frequencies; float if only one frequency, and
256            list otherwise.
257
258        """
259
260        # Range if two frequencies specified
261        if self.butterworth_filter_settings["cutoff2"] is not None:
262            cutoff_freq = [
263                self.butterworth_filter_settings["cutoff1"],
264                self.butterworth_filter_settings["cutoff2"],
265            ]
266        else:
267            cutoff_freq = self.butterworth_filter_settings["cutoff1"]
268
269        return cutoff_freq

285    def add_remove_mains(self, freq_remove: int = 50, n_win_avg: int = 51):
286        """
287        Add settings for removing mains noise. Settings will be passed to EMGDataRaw
288        method _remove_mains via the preprocess method.
289
290        Parameters
291        ----------
292        freq_remove : int, optional
293            The frequency to remove in hertz. The default is 50.
294        n_win_avg : int, optional
295            The number of windows to average to compute the mains noise signal.
296            The default is 51.
297
298        Raises
299        ------
300        ValueError
301            Raised if n_win_avg is not odd.
302
303        Returns
304        -------
305        None.
306
307        """
308
309        # Check that number of windows used to average noise is odd.
310        if n_win_avg % 2 == 0:
311            raise ValueError(
312                "The number of windows used to estimate the noise signal, n_win_avg, "
313                "must be odd."
314            )
315
316        # Save settings
317        self.remove_mains = True
318        self.remove_mains_settings = {
319            "freq_remove": freq_remove,
320            "n_win_avg": n_win_avg,
321        }
322
323    def remove_remove_mains(self):
324        """
325        Sets "remove mains" to False (mains noise will not be removed) and removes
326        corresponding parameters (in remove_mains_settings dictionary) from the
327        preprocessing settings.
328
329        Returns
330        -------
331        None.
332
333        """
334        self.remove_mains = False
335        self.remove_mains_settings = {}
